\subsection{Phân loại và ưu tiên tính năng}
\label{sec:feature-prioritization}

\begin{guide}
\guidepart{Nội dung}
\begin{compactitem}
    \item Danh sách tính năng ứng viên của 4 phân hệ (Video Streaming, Music Listening, News Reading, Minigames) và các tính năng dùng chung (tài khoản, tìm kiếm, thông báo, thanh toán).
    \item Đánh giá mỗi tính năng theo Impact (giá trị cho người dùng và cho kinh doanh) và Effort (công sức phát triển), rồi xếp vào 4 nhóm: Quick Wins, Major Projects, Fill-ins, Thankless Tasks.
\end{compactitem}
\guidepart{Câu hỏi cần trả lời}
\begin{compactitem}
    \item Impact và Effort được chấm theo thang điểm nào, ai chấm và dựa trên cơ sở gì?
    \item Những tính năng nào thuộc nhóm Quick Wins và sẽ được làm trước?
    \item Những tính năng nào thuộc nhóm Thankless Tasks, và vì sao bị loại hoặc hoãn?
\end{compactitem}
\guidepart{Hình/bảng nên có}
Ma trận Impact vs. Effort $2 \times 2$, bảng chấm điểm từng tính năng.
\end{guide}

Nội dung dưới đây là \textbf{bản tổng hợp tính năng ứng viên (brainstorm)} dành cho buổi thảo luận của nhóm: chủ đích là mở hết các hướng có thể nghĩ tới, ghi lại cả các biến thể và ý tưởng phụ, thay vì chốt sớm một hướng duy nhất. Impact và Effort của từng tính năng là \textit{đề xuất} của người soạn để nhóm thảo luận và chốt lại; bảng~\ref{tab:feature-pool} không phải kết luận.

\textbf{Thang điểm đề xuất.} Cả hai trục cùng chấm theo thang 1--5:

\begin{itemize}
    \item \textbf{Impact:} 1 = gần như không ai dùng; 3 = hữu ích rõ ràng với một nhóm người dùng; 5 = trụ cột của lý do chọn ứng dụng, gắn trực tiếp pain point ở mục~\ref{sec:problem-statement} hoặc một nhân tố Winning Zone trong bảng~\ref{tab:winning-zone}.
    \item \textbf{Effort:} 1 = vài ngày của một người; 3 = khoảng một tuần của một người hoặc ngang tương đương; 5 = rủi ro kỹ thuật cao (media engine, xử lý audio nền, bản quyền, ML).
\end{itemize}

Từ hai số này, mỗi tính năng được xếp vào một nhóm của ma trận Impact $\times$ Effort: \textbf{Quick Wins} (Impact cao, Effort thấp~--~làm trước), \textbf{Major Projects} (Impact cao, Effort cao~--~xếp lịch và cân nhắc kỹ), \textbf{Fill-ins} (Impact thấp, Effort thấp~--~làm khi dư thời gian), \textbf{Thankless Tasks} (Impact thấp, Effort cao~--~loại hoặc hoãn, ghi rõ lý do).
