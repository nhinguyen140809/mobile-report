\subsection{Unique Selling Point và Winning Zone}
\label{sec:usp-winning-zone}

\begin{guide}
\guidepart{Nội dung}
\begin{compactitem}
    \item Sơ đồ Venn 3 vòng: Nhu cầu người dùng (từ mục~\ref{sec:market-research} và~\ref{sec:user-personas}), Thế mạnh sản phẩm, Lỗ hổng của đối thủ. Giao điểm của cả ba là Winning Zone.
    \item Phát biểu USP trong 1--2 câu cho ứng dụng tích hợp 4 dịch vụ Video, Music, News, Minigames.
    \item Những vùng không nên cạnh tranh: đối thủ đã làm rất tốt hoặc người dùng không thực sự cần.
\end{compactitem}
\guidepart{Câu hỏi cần trả lời}
\begin{compactitem}
    \item Ứng dụng làm được điều gì mà một ứng dụng chuyên biệt (chỉ video, chỉ nhạc) không làm được?
    \item Mỗi yếu tố trong Winning Zone dựa trên bằng chứng nào (kết quả khảo sát, phân tích đối thủ)?
    \item USP được thể hiện qua tính năng cụ thể nào trong phạm vi MVP (mục~\ref{sec:mvp-scope})?
\end{compactitem}
\guidepart{Hình/bảng nên có}
Sơ đồ Venn xác định Winning Zone.
\end{guide}
